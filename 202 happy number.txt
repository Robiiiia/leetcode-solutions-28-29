282 happy number

import java.util.HashSet;
import java.util.Set;

class Solution {
    public boolean isHappy(int n) {
        // Создаём множество, чтобы хранить уже встреченные числа
        Set<Integer> seen = new HashSet<>();

        // Продолжаем вычисления, пока число не станет равным 1
        while (n != 1) {

            // Если число уже встречалось, значит, мы попали в цикл
            if (seen.contains(n)) {
                return false;
            }

            // Запоминаем текущее число, чтобы не повторять его бесконечно
            seen.add(n);

            // Переменная для хранения суммы квадратов цифр
            int sum = 0;

            // Обрабатываем все цифры числа по одной
            while (n > 0) {

                // Получаем последнюю цифру числа
                int digit = n % 10;

                // Возводим цифру в квадрат и добавляем к сумме
                sum += digit * digit;

                // Убираем последнюю цифру из числа
                n = n / 10;
            }

            // Заменяем число суммой квадратов его цифр
            n = sum;
        }

        // Если дошли сюда, число стало равным 1
        return true;
    }
}

class Solution {
    public boolean isPowerOfTwo(int n) {

        // Ноль и отрицательные числа не являются степенями двойки
        if (n <= 0) {
            return false;
        }

        // Продолжаем делить число на 2, пока оно делится без остатка
        while (n % 2 == 0) {

            // Делим число на 2
            n = n / 2;
        }

        // Если в конце осталось 1, исходное число было степенью двойки
        return n == 1;
    }
}
